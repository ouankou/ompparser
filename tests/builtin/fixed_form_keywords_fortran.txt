//Fixed-form source drops every blank in a directive line, so adjacent
//keywords arrive as one run. Each case must parse like its blank-separated
//spelling, while keyword-prefixed identifiers inside parentheses stay
//identifiers.

!$omp targetmap(tofrom:a,b)
PASS: !$omp target map(tofrom : a, b)

!$omp targetmap(a,b)
PASS: !$omp target map(a, b)

!$omp targetteamsdistributeparalleldosimd
PASS: !$omp target teams distribute parallel do simd

!$omp targetdatamap(to:a)
PASS: !$omp target data map(to : a)

!$omp targetenterdatamap(to:a)
PASS: !$omp target enter data map(to : a)

!$omp targetexitdatamap(from:a)
PASS: !$omp target exit data map(from : a)

!$omp targetupdateto(a)
PASS: !$omp target update to (a)

!$omp endtargetdata
PASS: !$omp end target data

!$omp endtargetteamsdistribute
PASS: !$omp end target teams distribute

!$omp targetdevice(1)map(to:a)
PASS: !$omp target device (1) map(to : a)

!$omp targetdatause_device_ptr(p)if(targetdata:c)
PASS: !$omp target data use_device_ptr(p) if (target data: c)

!$omp declaretargetenter(f)device_type(nohost)
PASS: !$omp declare target enter(f) device_type (nohost)

!$omp allocatorsallocate(allocator(omp_high_bw_mem_alloc):a)
PASS: !$omp allocators allocate (allocator(omp_high_bw_mem_alloc): a)

!$omp allocate(x)allocator(omp_default_mem_alloc)
PASS: !$omp allocate (x) allocator (omp_default_mem_alloc)

!$omp target_datamap(to:a)
PASS: !$omp target_data map(to : a)

!$omp target_enter_datamap(to:a)
PASS: !$omp target enter data map(to : a)

!$omp target_exit_datamap(from:a)
PASS: !$omp target exit data map(from : a)

!$omp metadirectivewhen(user={condition(n>10)}:targetmap(to:a))otherwise(targetteamsdistribute)

!$omp targetmap(to:target_function,data1,device_compute,cgroup_alloc,all_x,host_x)
PASS: !$omp target map(to : target_function, data1, device_compute, cgroup_alloc, all_x, host_x)

!$omp targetfirstprivate(target_x)is_device_ptr(data1)

!$omp flush(target_x,data1)
